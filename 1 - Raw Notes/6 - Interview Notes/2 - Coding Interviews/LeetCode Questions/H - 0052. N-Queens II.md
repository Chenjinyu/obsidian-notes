**Input:** n = 1
**Output:** 1

**Constraints:**
- `1 <= n <= 9`

```python
class Solution:
    def totalNQueens(self, n: int) -> int:
        ans = 0

        cols = set() 
        positive_diagonal = set() # row + col
        negative_disgonal = set() # row -col
        
        def backtrack(row):
            if row == n:
                # for looped all rows and all cols
                nonlocal ans
                ans += 1

            for col in range(n):
                if (
                    col in cols
                    or (row + col) in positive_diagonal
                    or (row - col) in negative_disgonal
                ):
                    # the (row, col) on the not allowed position. 
                    # because the two queens cannot stay in the same row, col, and positive and negative diagonals.
                    continue 

                # place queen
                cols.add(col)
                positive_diagonal.add(row + col) 
                negative_disgonal.add(row - col) 

                backtrack(row + 1)

                cols.remove(col)
                positive_diagonal.remove(row + col) 
                negative_disgonal.remove(row - col) 
            

        backtrack(0)
        return ans
```

```python
class Solution:
    def totalNQueens(self, n):
        def backtrack(row, diagonals, anti_diagonals, cols):
            # Base case - N queens have been placed
            if row == n:
                return 1

            solutions = 0
            for col in range(n):
                curr_diagonal = row - col
                curr_anti_diagonal = row + col
                # If the queen is not placeable
                if (col in cols 
                      or curr_diagonal in diagonals 
                      or curr_anti_diagonal in anti_diagonals):
                    continue

                # "Add" the queen to the board
                cols.add(col)
                diagonals.add(curr_diagonal)
                anti_diagonals.add(curr_anti_diagonal)

                # Move on to the next row with the updated board state
                solutions += backtrack(row + 1, diagonals, anti_diagonals, cols)

                # "Remove" the queen from the board since we have already
                # explored all valid paths using the above function call
                cols.remove(col)
                diagonals.remove(curr_diagonal)
                anti_diagonals.remove(curr_anti_diagonal)

            return solutions

        return backtrack(0, set(), set(), set())
```
